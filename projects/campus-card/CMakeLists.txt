cmake_minimum_required(VERSION 3.16)
project(CampusCard VERSION 0.1.1 LANGUAGES CXX)
include(CTest)
option(CAMPUS_BUILD_GUI "Build the Qt Widgets desktop application" OFF)
add_library(campus_core src/card_service.cpp)
target_include_directories(campus_core PUBLIC include)
set_target_properties(campus_core PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(campus_core PRIVATE /W4)
else()
  target_compile_options(campus_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(campus_cli src/main.cpp)
target_link_libraries(campus_cli PRIVATE campus_core)
set_target_properties(campus_cli PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(BUILD_TESTING)
  add_executable(campus_tests tests/test_card_service.cpp)
  target_link_libraries(campus_tests PRIVATE campus_core)
  set_target_properties(campus_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  add_test(NAME campus_domain COMMAND campus_tests)
endif()
if(CAMPUS_BUILD_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  add_executable(campus_gui gui/main.cpp)
  target_link_libraries(campus_gui PRIVATE campus_core Qt6::Widgets)
  set_target_properties(campus_gui PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  if(BUILD_TESTING)
    add_test(NAME campus_gui_smoke COMMAND campus_gui --smoke-test)
    set_tests_properties(campus_gui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
  endif()
endif()
install(TARGETS campus_cli RUNTIME DESTINATION bin)
