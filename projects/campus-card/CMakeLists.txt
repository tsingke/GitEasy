cmake_minimum_required(VERSION 3.16)
project(CampusCard VERSION 0.1.1 LANGUAGES CXX)
include(CTest)
option(CAMPUS_BUILD_GUI "Build the Qt Widgets desktop application" OFF)

# 每个目标明确语言标准；Qt的标准要求不向领域库反向传播。
function(campus_target target standard)
  set_target_properties(${target} PROPERTIES CXX_STANDARD ${standard} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()

add_library(campus_domain src/money.cpp src/business_date.cpp src/card_holder.cpp src/ledger_export.cpp)
target_include_directories(campus_domain PUBLIC include)
campus_target(campus_domain 11)
add_library(campus_core src/card_service.cpp)
target_link_libraries(campus_core PUBLIC campus_domain)
campus_target(campus_core 11)
add_executable(campus_cli src/main.cpp)
target_link_libraries(campus_cli PRIVATE campus_core)
campus_target(campus_cli 11)

if(BUILD_TESTING)
  function(campus_test target source dependency)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE ${dependency})
    campus_target(${target} 11)
    add_test(NAME ${target} COMMAND ${target})
  endfunction()
  campus_test(campus_tests tests/test_card_service.cpp campus_core)
  campus_test(campus_money tests/test_money.cpp campus_domain)
  campus_test(campus_calendar tests/test_business_date.cpp campus_domain)
  campus_test(campus_export tests/test_ledger_export.cpp campus_domain)
  campus_test(campus_clock tests/test_service_clock.cpp campus_core)
endif()

if(CAMPUS_BUILD_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  add_library(campus_widgets gui/campus_window.cpp gui/smoke_scenario.cpp)
  target_link_libraries(campus_widgets PUBLIC campus_core Qt6::Widgets)
  target_include_directories(campus_widgets PUBLIC gui)
  campus_target(campus_widgets 17)
  add_executable(campus_gui gui/main.cpp)
  target_link_libraries(campus_gui PRIVATE campus_widgets)
  campus_target(campus_gui 17)
  if(BUILD_TESTING)
    add_test(NAME campus_gui_smoke COMMAND campus_gui --smoke-test)
    set_tests_properties(campus_gui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
  endif()
endif()
install(TARGETS campus_cli RUNTIME DESTINATION bin)
